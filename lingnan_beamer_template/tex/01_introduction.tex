        \node[anchor=west] at (3.12,0.58) {\scriptsize food};
      \end{tikzpicture}
      }

      \vspace{0.10cm}
      {\scriptsize
      \textbf{Engel law:}
      \[
        \eta_{\mathrm{food}}<1,\qquad
        \eta_{\mathrm{manufacturing}},\,\eta_{\mathrm{services}}\ge 1 .
      \]
      }
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.28cm}
        \item \focus{Matsuyama (1992)}

          Stone--Geary demand $+$ learning-by-doing
          $\Longrightarrow$ closed vs. open economy outcomes.
        \item \focus{Echevarria (1997); Laitner (2000)}

          Income-elastic sectoral demand $\Longrightarrow$ reallocation.
        \item \focus{Cost}

          Specific, sometimes \emph{ad hoc}, preference restrictions.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

        \draw[arr] (subsistence) -- (demand);
        \draw[arr] (demand) -- (bgp);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
      \large
      \begin{itemize}
        \setlength{\itemsep}{0.36cm}
        \item \focus{Breakthrough:} Kuznets-style reallocation can coexist
          with Kaldor-style aggregate ratios.
        \item \focus{Cost:} subsistence demand parameters must be tied to
          sectoral productivity growth.
        \item \focus{Problem:} preference parameters and technology
          parameters are no longer independent.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Structural Change Mechanism}

\begin{frame}[label=employmentshares]{Employment Shares}
  \small
  \begin{block}{Consumption-only sectors}
    \begin{equation*}
      \hyperlink{appendixemploymentshares}{%
      n_i=\frac{x_i}{X}\left(\frac{c}{y}\right),
      \qquad i\neq m .
      \qquad (13)}
    \end{equation*}
  \end{block}

  \begin{alertblock}{Manufacturing}
    \begin{equation*}
      \hyperlink{appendixemploymentshares}{%
      n_m=\frac{x_m}{X}\left(\frac{c}{y}\right)
      +\underbrace{\left(1-\frac{c}{y}\right)}_{\text{savings rate}} .
      \qquad (14)}
    \end{equation*}
  \end{alertblock}

  \vspace{0.10cm}
  \begin{itemize}
    \item $\frac{x_i}{X}$ allocates consumption demand across final goods.
    \item $\frac{c}{y}$ is the labor share needed for total consumption.
    \item Manufacturing also supplies investment demand, captured by $1-c/y$.
  \end{itemize}
\end{frame}

\begin{frame}[label=relativeemployment]{Proposition 1: Relative Employment and Relative Prices}
  \footnotesize
  \begin{block}{Relative employment growth}
    \begin{equation*}
      \hyperlink{appendixrelative}{%
      \frac{\dot n_i}{n_i}-\frac{\dot n_j}{n_j}
      =(1-\varepsilon)(\gamma_j-\gamma_i),\qquad i,j\neq m .
      \qquad (15)}
    \end{equation*}
  \end{block}

  \begin{block}{Relative price growth}
    \begin{equation*}
      \frac{\dot p_i}{p_i}-\frac{\dot p_j}{p_j}
      =\gamma_j-\gamma_i,\qquad i,j\neq m .
      \qquad (16)
    \end{equation*}
    \vspace{-0.28cm}
    \begin{center}
      \scriptsize
      \(\displaystyle
      \left(\frac{p_i}{p_j}=\frac{A_j}{A_i}\right)
      \)
    \end{center}
    \vspace{-0.16cm}
  \end{block}

  \begin{alertblock}{Direct implication}
    \begin{equation*}
      \hyperlink{appendixrelative}{%
      \frac{\dot n_i}{n_i}-\frac{\dot n_j}{n_j}
      =(1-\varepsilon)\left(
      \frac{\dot p_i}{p_i}-\frac{\dot p_j}{p_j}\right).
      \qquad (17)}
    \end{equation*}
  \end{alertblock}
\end{frame}

\begin{frame}[label=individualdynamics]{Individual Employment Dynamics}
  \footnotesize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  Let
  \[
    \gammabar\equiv\sum_{i=1}^m\frac{x_i}{X}\gamma_i .
  \]

  \textbf{\textit{Consumption sectors:}} each sector follows a common intercept
  plus a TFP-growth term.
  \[
    \hyperlink{appendixindividual}{%
    \frac{\dot n_i}{n_i}
    =
    \frac{\dot{(c/y)}}{c/y}
    +(1-\varepsilon)(\gammabar-\gamma_i),
    \qquad i\neq m .
    \qquad (18)}
  \]

  \vspace{0.04cm}
  \textbf{\textit{Manufacturing:}} employment combines consumption-good
  production and capital-good production.
  \[
    \hyperlink{appendixmanufacturing}{%
    \begin{aligned}
      \frac{\dot n_m}{n_m}
      &=
      \left[
      \frac{\dot{(c/y)}}{c/y}
      +(1-\varepsilon)(\gammabar-\gamma_m)
      \right]\frac{(c/y)(x_m/X)}{n_m}
      -\frac{\dot{(c/y)}}{1-c/y}\frac{1-c/y}{n_m}.
    \end{aligned}
    \qquad (19)}
  \]
\end{frame}

\begin{frame}{Proposition 2: When Do Employment Shares Change?}
  \footnotesize
  \begin{center}
    {\color{black}%
    \(\displaystyle
      \frac{\dot n_i}{n_i}
      =
      \frac{\dot{(c/y)}}{c/y}
      +(1-\varepsilon)(\gammabar-\gamma_i).
      \qquad i\neq m .
      \qquad (18)
    \)}
  \end{center}

  \vspace{0.06cm}
  \begin{tabularx}{\textwidth}{@{}p{0.30\textwidth}X X@{}}
    \toprule
    Case & Condition for structural change & Type of reallocation \\
    \midrule
    Equal sectoral TFP growth
      & $\dot c/c\neq \dot y/y$
      & Between the aggregate consumption sector and the manufacturing capital-goods role. \\
    Different sectoral TFP growth on a balanced path
      & $\varepsilon\neq 1$ and at least one $\gamma_i\neq\gamma_m$
      & Across consumption sectors with different TFP growth rates. \\
    \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}{The Sign of $1-\varepsilon$ Determines the Direction}
  \footnotesize
  \begin{center}
    {\color{black}
    \(\displaystyle
      \gamma_i<\gamma_j
      \quad\Longrightarrow\quad
      \gamma_j-\gamma_i>0
    \)}

    \vspace{0.06cm}
    {\color{black}%
    \(\displaystyle
      \frac{\dot n_i}{n_i}
      -
      \frac{\dot n_j}{n_j}
      =
      (1-\varepsilon)(\gamma_j-\gamma_i),
      \qquad i,j\neq m .
      \qquad (17)
    \)}
  \end{center}

  \vspace{0.05cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.32\textwidth}
      \begin{alertblock}{$\varepsilon<1$}
        \begin{itemize}
          \item Goods are poor substitutes.
          \item Demand is price inelastic.
          \item Labor moves toward slow-TFP-growth sectors.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{$\varepsilon=1$}
        \begin{itemize}
          \item Cobb-Douglas aggregator.
          \item Expenditure shares are constant.
          \item TFP gaps change prices but not employment shares.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{$\varepsilon>1$}
        \begin{itemize}
          \item Goods are strong substitutes.
          \item Demand is price elastic.
          \item Labor moves toward fast-TFP-growth sectors.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.18cm}
  \begin{center}
    \begin{tikzpicture}[x=1cm,y=1cm]
      \draw[-{Latex[length=2mm]},thick] (0,0) -- (8.8,0);
      \node[anchor=east] at (0,0) {\scriptsize slow TFP};
      \node[anchor=west] at (8.8,0) {\scriptsize fast TFP};
      \draw[darkred!80!black,very thick,-{Latex[length=2mm]}] (5.9,0.45) -- (2.5,0.45)
        node[midway,above] {\scriptsize $\varepsilon<1$};
      \draw[black!70,very thick,-{Latex[length=2mm]}] (2.9,-0.45) -- (6.3,-0.45)
        node[midway,below] {\scriptsize $\varepsilon>1$};
    \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}[label=nominalshares]{Nominal Shares Move More Than Real Shares}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{block}{Nominal side}
        \scriptsize
        \textbf{\textit{Nominal output shares:}}
        \[
          {\color{black}
          \frac{p_iF^i}{p_my}=n_i.}
        \]

        \textbf{\textit{Nominal consumption shares:}}
        \[
          {\color{black}
          \frac{p_ic_i}{\sum_{j=1}^m p_jc_j}=\frac{x_i}{X}.}
        \]

        Both nominal shares inherit the same relative reallocation force:
        \[
          {\color{black}
          \frac{d}{dt}\log\frac{n_i}{n_j}
          =
          \frac{d}{dt}\log\frac{x_i/X}{x_j/X}
          =
          (1-\varepsilon)(\gamma_j-\gamma_i).}
        \]
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{alertblock}{Real side}
        \scriptsize
        \textbf{\textit{Real consumption shares:}}
        \begin{equation*}
          {\color{black}
          \frac{\dot c_i}{c_i}-\frac{\dot c_j}{c_j}
          =\varepsilon(\gamma_i-\gamma_j)
          \qquad (20)}
        \end{equation*}
        Small $\varepsilon$ dampens changes in real consumption shares.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

  \vspace{0.25cm}
  \begin{alertblock}{Main intuition for $\varepsilon<1$}
    Productivity growth makes a sector cheaper. But if goods are poor
    substitutes, demand does not rise enough to absorb all extra output, so
    labor leaves the fast-growing sector.
  \end{alertblock}
\end{frame}

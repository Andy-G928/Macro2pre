\begin{frame}{Present-Value Hamiltonian}
  \scriptsize
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}
  \setlength{\jot}{4pt}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      {\bfseries Hamiltonian}
      \vspace{-0.08cm}
      \begin{equation*}
        \begin{aligned}
          \mathcal H^P
          =&\, e^{-\rho t}v(c_1,\ldots,c_m)\\
          &+q\!\left[
            F^m(n_mk_m,n_m)-c_m-(\delta+\nu)k
          \right]\\
          &+\sum_{i=1}^{m-1}\mu_i\!\left[
            F^i(n_ik_i,n_i)-c_i
          \right]\\
          &+\lambda\!\left(1-\sum_{i=1}^m n_i\right)\\
          &+\eta\!\left(k-\sum_{i=1}^m n_i k_i\right).
        \end{aligned}
      \end{equation*}
    \end{column}
    \begin{column}{0.37\textwidth}
      \textbf{Optimization logic}
      \vspace{0.04cm}
      \begin{itemize}
        \setlength\itemsep{0.06cm}
        \item $\partial\mathcal H^P/\partial c_i=0$ and
          $\partial\mathcal H^P/\partial c_m=0$: identify the shadow
          values of consumption goods.
        \item $\partial\mathcal H^P/\partial k_i=0$: equalize value
          marginal products of capital.
        \item $\partial\mathcal H^P/\partial n_i=0$: equalize value
          marginal products of labor.
        \item $\dot q=-\partial\mathcal H^P/\partial k$: obtain the
          dynamic efficiency condition.
      \end{itemize}

      \vspace{0.04cm}
      \textbf{Transversality}
      \vspace{-0.10cm}
      \begin{equation*}
        \lim_{t\to\infty}
        k(t)\exp\!\left[-\int_0^t
        \bigl(F_K^m(\tau)-\delta-\nu\bigr)d\tau\right]=0 .
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[label=efficiencycondition]{Efficiency Condition}
  \scriptsize
  \textbf{\textit{Static efficiency conditions:}} factor reallocation stops only when
  value marginal products are equal across sectors.

  \vspace{0.10cm}
  \begin{center}
    \hyperlink{appendixstatic}{%
      \(\displaystyle
      \frac{v_i}{v_m}
      =
      \frac{F_K^m}{F_K^i}
      =
      \frac{F_N^m}{F_N^i}
      \quad \forall i.
      \qquad (5)
      \)}
  \end{center}

  \vspace{0.14cm}
  \textbf{\textit{Dynamic efficiency condition:}} the marginal utility of the
  manufacturing good evolves with the net return to capital.

  \vspace{0.10cm}
  \begin{center}
    \hyperlink{appendixdynamic}{%
      \(\displaystyle
      -\frac{\dot v_m}{v_m}
      =
      F_K^m-(\delta+\rho+\nu).
      \qquad (6)
      \)}
  \end{center}
\end{frame}

\begin{frame}[label=tfpprices]{TFP Gaps Become Relative Prices}
  \scriptsize
  \begin{center}
    \hyperlink{appendixstatic}{%
      \(\displaystyle
      \frac{v_i}{v_m}
      =
      \frac{F_K^m}{F_K^i}
      =
      \frac{F_N^m}{F_N^i}
      \quad \forall i.
      \qquad (5)
      \)}
  \end{center}

  \begin{block}{Sectoral technologies}
    \vspace{-0.16cm}
    \begin{equation*}
      F^i=A_i F(n_i k_i,n_i),\qquad
      \dot{A}_i/A_i=\gamma_i .
    \end{equation*}
    \vspace{-0.34cm}
  \end{block}

  \vspace{0.04cm}
  \begin{equation*}
    \begin{aligned}
      \frac{F_N^i}{F_K^i}=\frac{F_N^m}{F_K^m}
      &\Longrightarrow
      \frac{f(k_i)}{f'(k_i)}-k_i
      =
      \frac{f(k_m)}{f'(k_m)}-k_m
      \Longrightarrow k_i=k,\\[0.08cm]
      \frac{v_i}{v_m}=\frac{F_K^m}{F_K^i}
      &\Longrightarrow
      \frac{v_i}{v_m}
      =
      \frac{A_m f'(k)}{A_i f'(k)}
      =
      \frac{A_m}{A_i}.
    \end{aligned}
  \end{equation*}

  \begin{alertblock}{Static efficiency implies}
    \vspace{-0.18cm}
    \begin{center}
      \hyperlink{appendixtfp}{%
        \(\displaystyle
        k_i=k,\qquad
        \frac{p_i}{p_m}=\frac{v_i}{v_m}=\frac{A_m}{A_i},
        \quad \forall i.
        \qquad (8)
        \)}
    \end{center}
    \vspace{-0.28cm}
  \end{alertblock}
\end{frame}

\begin{frame}{CES Preferences}
  \small
  \begin{equation*}
    v(c_1,\ldots,c_m)=\frac{\phi(c_1,\ldots,c_m)^{1-\theta}-1}{1-\theta},
  \end{equation*}
  \begin{equation*}
    \phi=\left(\sum_{i=1}^m \omega_i c_i^{(\varepsilon-1)/\varepsilon}\right)^{\varepsilon/(\varepsilon-1)} .
  \end{equation*}

  \vspace{0.12cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{$\varepsilon$: substitution across goods}
        Determines whether rising relative prices increase or reduce a
        sector's expenditure and employment shares.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{$1/\theta$: intertemporal substitution}
        Determines whether structural change can coexist with a balanced
        aggregate path.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[label=expenditureratio]{Expenditure Ratio}
  \scriptsize
  \begin{itemize}
    \setlength\itemsep{0.16cm}
    \item CES optimality gives
      $
      c_i/c_m=(\omega_i/\omega_m)^\varepsilon(p_i/p_m)^{-\varepsilon}
      $
      \(\bigl(p_i/p_m=v_i/v_m\bigr)\).
    \item Multiplying by $p_i/p_m$ converts relative quantities into
      relative expenditures; substituting $p_i/p_m=A_m/A_i$ gives $x_i$.
    \item \textbf{Consumption expenditure of sector $i$ relative to manufacturing}
  \end{itemize}

  \vspace{0.02cm}
  \begin{equation*}
    \hyperlink{appendixexpenditure}{%
    \frac{p_i c_i}{p_m c_m}
      =
      \left(\frac{\omega_i}{\omega_m}\right)^\varepsilon
      \left(\frac{A_m}{A_i}\right)^{1-\varepsilon}
      \equiv x_i .
    \qquad (10)}
  \end{equation*}

  \vspace{0.08cm}
  \begin{center}
    \begin{tikzpicture}[x=1cm,y=1cm]
      \node[draw=darkred!80!black,rounded corners,align=center,inner sep=4pt]

      \node[draw=black!55,rounded corners,align=center,inner sep=4pt]

      \node[draw=darkred!80!black,rounded corners,align=center,inner sep=4pt]

\end{frame}

\begin{frame}{Aggregate Variables in Manufacturing Units}
  \small
  \begin{equation*}
    c\equiv\sum_{i=1}^m \frac{p_i}{p_m}c_i,\qquad
    y\equiv\sum_{i=1}^m \frac{p_i}{p_m}F^i .
  \end{equation*}

  \vspace{0.12cm}
  \begin{block}{Two useful simplifications}
    \begin{equation*}
      c=c_m X,\qquad y=A_m F(k,1),\qquad
      X\equiv\sum_{i=1}^m x_i .
    \end{equation*}
  \end{block}

  \vspace{0.12cm}
  \begin{itemize}
    \item $x_i/X$ is sector $i$'s consumption expenditure share.
    \item $c/y$ is the economy-wide employment share used for consumption goods.
    \item $1-c/y$ is the manufacturing labor share used to produce capital goods.
  \end{itemize}
\end{frame}
